    <script>

        const GOOGLE_SHEET_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbyNkKZU2MBNYo4FiZGQCxVqB9_nLeQuyQncdo37zD5u7gOePQ0a95vmCaeTKj8ZrOm6Jg/exec";

        const LINE_OA_ID = "@332pxzud";

        const LIFF_ID = "2010166992-Ms8DVgUg";

 

        const statusEl   = document.getElementById('status');

        const subTextEl  = document.getElementById('sub-text');

        const btn        = document.getElementById('send-btn');

        const profileImg = document.getElementById('profile-img');

        const surveyBox  = document.getElementById('survey-box');

        const errMsg     = document.getElementById('err-msg');

 

        let currentProfile = null;

 

        // --- ระบบดึงค่าพารามิเตอร์ (Tracking Source) ---

        const urlParams = new URLSearchParams(window.location.search);

        let trackingSource = urlParams.get('source');

        if (trackingSource) {

            sessionStorage.setItem('liff_source', trackingSource);

        } else {

            trackingSource = sessionStorage.getItem('liff_source') || 'Organic / No Ref';

        }

        // ---------------------------------------------

 

        async function main() {

            try {

                await liff.init({ liffId: "2010166992-Ms8DVgUg"});

 

                if (!liff.isLoggedIn()) {

                    liff.login();

                    return;

                }

 

                const friendship = await liff.getFriendship();

 

                if (friendship.friendFlag) {

                    // ---------- STEP 1 : เป็นเพื่อนแล้ว ----------

                    currentProfile = await liff.getProfile();

                    profileImg.src = currentProfile.pictureUrl;

                    profileImg.style.display = "block";

 

                    // บันทึกการเป็นเพื่อนทันที (กันเคส user ปิดหน้าก่อนกรอกฟอร์ม = ยังได้ Lead)

                    saveFriendToSheet(currentProfile);

 

                    // ถ้าเคยกรอกฟอร์มรอบนี้แล้ว ข้ามไปหน้าแชทเลย

                    if (sessionStorage.getItem('survey_done') === '1') {

                        showChatStep();

                    } else {

                        showSurveyStep();

                    }

                } else {

                    // ---------- STEP 0 : ยังไม่เป็นเพื่อน ----------

                    statusEl.innerText = "ยังไม่ได้เพิ่มเพื่อน";

                    subTextEl.innerText = "กำลังเปิดหน้าเพิ่มเพื่อน...";

                    if (typeof liff.requestFriendship === 'function') {

                        await liff.requestFriendship();

                        location.reload();

                    } else {

                        // fallback: LIFF SDK บางเวอร์ชันไม่มี requestFriendship

                        window.location.href = `https://line.me/R/ti/p/${LINE_OA_ID}`;

                    }

                }

            } catch (err) {

                statusEl.innerText = "เกิดข้อผิดพลาด";

                subTextEl.innerText = err.message;

            }

        }

 

        // ================= STEP 2 : แสดงแบบสอบถาม =================

        function showSurveyStep() {

            statusEl.innerText = `สวัสดีคุณ ${currentProfile.displayName}`;

            subTextEl.innerText = "ยินดีที่ได้เป็นเพื่อนกันครับ! รบกวนตอบ 3 ข้อก่อนเริ่มแชทนะครับ";

 

            surveyBox.style.display = "block";

            btn.innerText = "ถัดไป";

            btn.style.display = "block";

            btn.onclick = submitSurvey;

        }

 

        async function submitSurvey() {

            const q1 = document.getElementById('q1');

            const q2 = document.getElementById('q2');

            const q3 = document.getElementById('q3');

 

            let valid = true;

            [q1, q2, q3].forEach(el => {

                if (!el.value) { el.classList.add('error'); valid = false; }

                else { el.classList.remove('error'); }

            });

 

            if (!valid) { errMsg.style.display = "block"; return; }

            errMsg.style.display = "none";

 

            btn.disabled = true;

            btn.innerText = "กำลังบันทึก...";

 

            // รอให้บันทึกเสร็จก่อน แล้วค่อยไปหน้าถัดไป (กันข้อมูลหายตอน redirect)

            await saveSurveyToSheet({

                interest: q1.value,

                grade: q2.value,

                timing: q3.value

            });

 

            sessionStorage.setItem('survey_done', '1');

            surveyBox.style.display = "none";

            btn.disabled = false;

            showChatStep();

        }

 

        // ================= STEP 3 : ปุ่มไปห้องแชท =================

        function showChatStep() {

            statusEl.innerText = "ขอบคุณครับ!";

            subTextEl.innerText = "กดปุ่มด้านล่างเพื่อเริ่มแชทกับเราได้เลย";

            btn.innerText = "กดเพื่อทักทาย AI หมาคับ";

            btn.style.display = "block";

            btn.onclick = redirectToOA;

        }

 

        function redirectToOA() {

            const message = "สวัสดีครับ AI หมาคับ! ยินดีที่ได้รู้จักครับ 🐶";

            const encodedMsg = encodeURIComponent(message);

            window.location.href = `https://line.me/R/oaMessage/${LINE_OA_ID}/?${encodedMsg}`;

        }

 

        // ================= ส่งข้อมูลเข้า Google Sheet =================

        // action = "add_friend" -> สร้างแถวใหม่

        function saveFriendToSheet(profile) {

            return postToSheet({

                action: "add_friend",

                userId: profile.userId,

                displayName: profile.displayName,

                pictureUrl: profile.pictureUrl,

                source: trackingSource

            });

        }

 

        // action = "survey" -> อัปเดตแถวเดิมของ userId นั้น

        function saveSurveyToSheet(answers) {

            return postToSheet({

                action: "survey",

                userId: currentProfile.userId,

                displayName: currentProfile.displayName,

                pictureUrl: currentProfile.pictureUrl,

                source: trackingSource,

                interest: answers.interest,

                grade: answers.grade,

                timing: answers.timing

            });

        }

 

        function postToSheet(data) {

            return fetch(GOOGLE_SHEET_WEB_APP_URL, {

                method: 'POST',

                body: JSON.stringify(data),

                headers: { 'Content-Type': 'text/plain;charset=utf-8' }

            }).catch(error => console.error("Sheet Error:", error));

        }

 

        main();
